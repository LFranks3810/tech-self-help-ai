document.getElementById('chat-form').addEventListener('submit', async function(e) { e.preventDefault(); const input = document.getElementById('chat-input'); const message = input.value.trim(); if (!message) return; const chatContainer = document.getElementById('chat-messages'); // Append User Message chatContainer.innerHTML += `
${message}
`; input.value = ''; chatContainer.scrollTop = chatContainer.scrollHeight; let botReply = "I'm here to help with SentriForge support! Let me check that for you."; try { const apiUrl = localStorage.getItem('sentriforge_api_url') || ''; const response = await fetch(`${apiUrl}/api/chat`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ message }) }); const data = await response.json(); if (data && data.reply) { botReply = data.reply; } } catch (err) { console.warn("API Fetch failed, using local backup"); if (message.toLowerCase().includes('password')) { botReply = "To reset your password, visit the self-service portal or submit an Access ticket."; } else if (message.toLowerCase().includes('vpn')) { botReply = "For VPN issues, verify your MFA token in Microsoft Authenticator and re-authenticate."; } } // Append Bot Response chatContainer.innerHTML += `
${botReply}
`; chatContainer.scrollTop = chatContainer.scrollHeight; });
